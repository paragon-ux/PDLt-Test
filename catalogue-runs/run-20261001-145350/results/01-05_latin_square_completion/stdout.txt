from copy import deepcopy

def solve_latin_square(grid):
    n = len(grid)
    # possible values for each cell
    possibilities = [[set(range(1, n+1)) if grid[r][c]==0 else {grid[r][c]} for c in range(n)] for r in range(n)]
    # initial propagation
    def propagate(poss):
        changed = True
        while changed:
            changed = False
            for r in range(n):
                for c in range(n):
                    if len(poss[r][c])==1:
                        val = next(iter(poss[r][c]))
                        # eliminate from row
                        for cc in range(n):
                            if cc!=c and val in poss[r][cc]:
                                poss[r][cc].remove(val)
                                changed = True
                        # eliminate from column
                        for rr in range(n):
                            if rr!=r and val in poss[rr][c]:
                                poss[rr][c].remove(val)
                                changed = True
        return poss
    possibilities = propagate(possibilities)
    # backtracking search
    def backtrack(poss):
        # find unsolved cell with fewest possibilities >1
        min_len = n+1
        target = None
        for r in range(n):
            for c in range(n):
                l = len(poss[r][c])
                if l>1 and l<min_len:
                    min_len = l
                    target = (r,c)
        if target is None:
            # all assigned
            return poss
        r,c = target
        for val in list(poss[r][c]):
            new_poss = deepcopy(poss)
            new_poss[r][c] = {val}
            new_poss = propagate(new_poss)
            # check for empty possibility -> dead end
            if any(len(new_poss[i][j])==0 for i in range(n) for j in range(n)):
                continue
            result = backtrack(new_poss)
            if result is not None:
                return result
        return None
    final = backtrack(possibilities)
    if final is None:
        return None
    return [[next(iter(final[r][c])) for c in range(n)] for r in range(n)]
