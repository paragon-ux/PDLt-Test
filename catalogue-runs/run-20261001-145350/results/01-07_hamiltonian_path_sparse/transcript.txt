from collections import defaultdict

def find_hamiltonian_path(edges, n):
    adj = defaultdict(list)
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
    visited = [False]*n
    path = []
    nodes_explored = 0
    def backtrack(node):
        nonlocal nodes_explored
        visited[node] = True
        path.append(node)
        nodes_explored += 1
        if len(path) == n:
            return True
        for nb in adj[node]:
            if not visited[nb]:
                if backtrack(nb):
                    return True
        visited[node] = False
        path.pop()
        return False
    for start in range(n):
        if backtrack(start):
            return path, nodes_explored
    return None, nodes_explored
