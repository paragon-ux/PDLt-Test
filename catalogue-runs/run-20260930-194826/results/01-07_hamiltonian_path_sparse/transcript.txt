N = 12

def backtrack(path, visited):
    if len(path) == N:
        return path
    last = path[-1]
    for neighbor in sorted(graph[last]):
        if neighbor not in visited:
            visited.add(neighbor)
            path.append(neighbor)
            result = backtrack(path, visited)
            if result:
                return result
            path.pop()
            visited.remove(neighbor)
    return None

solution = None
