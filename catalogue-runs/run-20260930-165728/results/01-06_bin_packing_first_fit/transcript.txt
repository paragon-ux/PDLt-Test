def check_packing(packing, capacity=10):
    all_items = []
    for bin_items in packing:
        assert sum(bin_items) <= capacity, f"Bin exceeds capacity: {bin_items}"
        all_items.extend(bin_items)
    expected = [5,5,5,3,3,3,7,7]
