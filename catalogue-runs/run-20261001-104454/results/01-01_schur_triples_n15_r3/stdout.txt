from itertools import combinations

# generate all possible triples (a,b,c) where a + b = c
triples = [(a, b, c) for c in L for a, b in combinations(L, 2) if a + b == c]

def exact_cover(elements, subsets):
    if not elements:
        return []
    # map each element to subsets containing it
    elem_to_subs = {e: [] for e in elements}
    for i, sub in enumerate(subsets):
        for e in sub:
            if e in elem_to_subs:
                elem_to_subs[e].append(i)
    # choose element with fewest covering subsets
    chosen_elem = min(elements, key=lambda e: len(elem_to_subs[e]))
    for sub_idx in elem_to_subs[chosen_elem]:
        sub = subsets[sub_idx]
        new_elements = elements - set(sub)
        new_subsets = [s for j, s in enumerate(subsets) if j != sub_idx and not set(s) & set(sub)]
        result = exact_cover(new_elements, new_subsets)
        if result is not None:
            return [sub] + result
    return None

solution = exact_cover(set(L), triples)
