def find_partition(numbers):
    numbers = list(numbers)
    s = set(numbers)
    triples = []
    for i, a in enumerate(numbers):
        for b in numbers[i+1:]:
            if a + b in s:
                c = a + b
                if c != a and c != b:
                    triples.append((a, b, c))
    elem_to_triples = {x: [] for x in numbers}
    for idx, (a, b, c) in enumerate(triples):
        elem_to_triples[a].append(idx)
        elem_to_triples[b].append(idx)
        elem_to_triples[c].append(idx)

    solution = []

    def dfs(remaining):
        if not remaining:
            return True
        # choose element with fewest options
        min_elem = None
        min_options = None
        for e in remaining:
            options = [ti for ti in elem_to_triples[e] if all(x in remaining for x in triples[ti])]
            if not options:
                return False
            if min_options is None or len(options) < len(min_options):
                min_elem = e
                min_options = options
        for ti in list(min_options):
            a, b, c = triples[ti]
            solution.append((a, b, c))
            new_remaining = remaining - {a, b, c}
            if dfs(new_remaining):
                return True
            solution.pop()
        return False

    if dfs(set(numbers)):
        return solution
    else:
        return None
