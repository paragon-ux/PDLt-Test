L = [71, 97, 54, 56, 44, 158, 45, 58, 82, 135, 113, 117, 86, 40, 12, 185, 63, 50, 111, 72, 90, 109, 61, 13, 92, 60, 91, 35, 114, 94, 105, 26, 15, 51, 102, 118, 76, 32, 108, 23, 22, 134, 83, 43, 64]

from itertools import combinations

# generate all possible triples (a, b, c) where a + b = c
triples = [(a, b, a + b) for a, b in combinations(L, 2) if a + b in L]

# map each number to the triples that contain it
from collections import defaultdict
triples_by_num = defaultdict(list)
for t in triples:
    a, b, c = t
    triples_by_num[a].append(t)
    triples_by_num[b].append(t)
    triples_by_num[c].append(t)

def search(available):
    if not available:
        return []
    x = min(available)
    for t in triples_by_num[x]:
        a, b, c = t
        if a in available and b in available and c in available:
            new_avail = available - {a, b, c}
            res = search(new_avail)
            if res is not None:
                return [t] + res
    return None

solution = search(set(L))
import json
if solution is None:
    witness = {
        'polarity': 'negative',
        'evidence': {'path': 'execution://witness'},
        'basis': 'proof',
        'argument': 'No valid partition exists.'
    }
else:
    partition = [[a, b, c] for a, b, c in solution]
    witness = {
        'polarity': 'positive',
        'evidence': {'path': 'execution://witness'},
        'data': {'partition': partition}
    }
print('WITNESS: ' + json.dumps(witness))
