'''\nExact cover solver for universe U = {1..9} with sets S1..S9.\nFinds all exact covers using Algorithm X (simple recursive implementation).\nAlso includes a self-contained test verifying that each solution covers U exactly once.\n'''\n\nfrom typing import List, Dict, Set\n\n# Define the universe and the collection of sets\nU: Set[int] = {1,2,3,4,5,6,7,8,9}\nSUBSETS: Dict[str, Set[int]] = {\n    'S1': {1,2,3},\n    'S2': {4,5,6},\n    'S3': {7,8,9},\n    'S4': {1,4,7},\n    'S5': {2,5,8},\n    'S6': {3,6,9},\n    'S7': {1,5,9},\n    'S8': {2,6,7},\n    'S9': {3,4,8},\n}\n\ndef solve_exact_cover(universe: Set[int], subsets: Dict[str, Set[int]]) -> List[List[str]]:\n    '''Return all exact covers as lists of subset names.'''
    solutions: List[List[str]] = []\n\n    # Build element -> subset name mapping for quick lookup\n    element_to_subsets: Dict[int, List[str]] = {e: [] for e in universe}\n    for name, elems in subsets.items():\n        for e in elems:\n            element_to_subsets[e].append(name)\n\n    def search(remaining: Set[int], partial: List[str]) -> None:\n        if not remaining:\n            solutions.append(partial.copy())\n            return\n        # Choose the element with fewest possible subsets (heuristic)\n        elem = min(remaining, key=lambda e: len(element_to_subsets[e]))\n        for subset_name in element_to_subsets[elem]:\n            subset = subsets[subset_name]\n            if not subset <= remaining:\n                continue\n            new_remaining = remaining - subset\n            partial.append(subset_name)\n            # Temporarily hide subsets that intersect with chosen subset\n            saved = {}\n            for e in subset:\n                saved[e] = element_to_subsets[e]\n                element_to_subsets[e] = []\n            search(new_remaining, partial)\n            partial.pop()\n            for e in subset:\n                element_to_subsets[e] = saved[e]\n\n    search(set(universe), [])\n    return solutions\n\ndef test_solutions(solutions: List[List[str]]) -> None:\n    '''Verify each solution covers every element of U exactly once.'''\n    for idx, sol in enumerate(solutions, 1):\n        covered: Set[int] = set()\n        for name in sol:\n            elems = SUBSETS[name]\n            assert covered.isdisjoint(elems), f'Overlap in solution #{idx}'\n            covered.update(elems)\n        assert covered == U, f'Incomplete cover in solution #{idx}'\n    print(f'All {len(solutions)} solution(s) passed the verification test.')\n\nif __name__ == '__main__':\n    all_solutions = solve_exact_cover(U, SUBSETS)\n    print('Exact covers (each list contains subset names):')\n    for sol in all_solutions:\n        print(sol)\n    test_solutions(all_solutions)
