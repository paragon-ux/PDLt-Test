# First Fit and optimal bin packing for Items = [5,5,5,3,3,3,7,7] with capacity C=10
# Implements First Fit algorithm, brute‑force optimal search, and a self‑test.
# All code uses only ASCII characters.

import itertools
from typing import List, Tuple

C = 10
ITEMS = [5, 5, 5, 3, 3, 3, 7, 7]

def first_fit(items: List[int]) -> List[List[int]]:
    bins: List[List[int]] = []
    for item in items:
        placed = False
        for b in bins:
            if sum(b) + item <= C:
                b.append(item)
                placed = True
                break
        if not placed:
            bins.append([item])
    return bins

def is_valid_packing(packing: List[List[int]]) -> bool:
    # every item used exactly once and no bin exceeds capacity
    all_items = [i for bin_ in packing for i in bin_]
    if sorted(all_items) != sorted(ITEMS):
        return False
    return all(sum(bin_) <= C for bin_ in packing)

def optimal_packing(items: List[int]) -> List[List[int]]:
    # brute‑force search over all partitions (limited size, feasible here)
    best: List[List[int]] = []
    n = len(items)
    # generate all assignments of items to bins numbered 0..n-1
    for k in range(1, n+1):  # number of bins used
        # each item assigned a bin index < k
        for assign in itertools.product(range(k), repeat=n):
            bins = [[] for _ in range(k)]
            for idx, bin_idx in enumerate(assign):
                bins[bin_idx].append(items[idx])
            # discard empty bins
            bins = [b for b in bins if b]
            if is_valid_packing(bins):
                best = bins
                return best
    return best

def self_test():
    ff = first_fit(ITEMS)
    opt = optimal_packing(ITEMS)
    assert is_valid_packing(ff), "First Fit packing invalid"
    assert is_valid_packing(opt), "Optimal packing invalid"
    print("First Fit packing:", ff)
    print("Optimal packing:", opt)
    print("Number of bins - First Fit:", len(ff))
    print("Number of bins - Optimal:", len(opt))
    gap = len(ff) - len(opt)
    print("Gap (FF bins - optimal bins):", gap)
    # Explanation of why order hurts First Fit
    print("Explanation: The arrival order places three 5s first, each opening a new bin. The subsequent 3s cannot fit into those bins because each already contains a 5 (5+3=8 fits, but the algorithm keeps items in the first bin that has space; after the three 5s, the first bin has space 5, second 5, third 5. The first 3 fits into the first bin (5+3=8), the second 3 fits into the second bin, the third 3 fits into the third bin. The later 7s each need a new bin because all existing bins are already at 8 or 5+3=8, leaving only 2 capacity. An optimal rearrangement would pair each 7 with a 3 (7+3=10) and pack the three 5s together (5+5=10) using only three bins.)
