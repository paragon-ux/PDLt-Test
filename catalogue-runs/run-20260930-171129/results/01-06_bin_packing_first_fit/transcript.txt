C = 10

def first_fit(items):
    bins = []
    for w in items:
        placed = False
        for b in bins:
            if sum(b) + w <= C:
                b.append(w)
                placed = True
                break
        if not placed:
            bins.append([w])
    return bins

def optimal_packing(items):
    # brute‑force search for minimal number of bins (small instance)
    best = None
    def backtrack(i, bins):
        nonlocal best
        # prune if already worse than best found
        if best is not None and len(bins) >= len(best):
            return
        if i == len(items):
            best = [list(b) for b in bins]
            return
        w = items[i]
        # try to place in existing bins
        for b in bins:
            if sum(b) + w <= C:
                b.append(w)
                backtrack(i + 1, bins)
                b.pop()
        # try a new bin
        bins.append([w])
        backtrack(i + 1, bins)
        bins.pop()
    backtrack(0, [])
    return best

ff = first_fit(items)
opt = optimal_packing(items)
gap = len(ff) - len(opt)
