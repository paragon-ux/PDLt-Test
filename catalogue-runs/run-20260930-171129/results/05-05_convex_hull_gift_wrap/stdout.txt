def orientation(p, q, r):
    """Return >0 if counter‑clockwise, <0 if clockwise, 0 if collinear."""
    return (q[0] - p[0]) * (r[1] - p[1]) - (q[1] - p[1]) * (r[0] - p[0])

def jarvis_march(points):
    """Return convex hull vertices in counter‑clockwise order, including all
    points that lie on the hull boundary. Duplicates are ignored in the output.
    """
    # Remove duplicate points for algorithm stability
    pts = list(set(points))
    n = len(pts)
    if n == 0:
        return []
    if n == 1:
        return pts
    # If all points are collinear, return them sorted along the line
    collinear = all(orientation(pts[0], pts[1], p) == 0 for p in pts[2:])
    if collinear:

        sorted_pts = sorted(pts, key=lambda p: (p[0], p[1]))
        # return from leftmost to rightmost and back to leftmost to keep CCW
        return sorted_pts
    # Find leftmost (lowest x, then lowest y) point – guaranteed to be hull start
    start = min(pts, key=lambda p: (p[0], p[1]))

    point_on_hull = start
    while True:
        hull.append(point_on_hull)
        # Choose a candidate that is most counter‑clockwise relative to point_on_hull
        candidate = None
        for p in pts:
            if p == point_on_hull:
                continue
            if candidate is None:
                candidate = p
                continue
            o = orientation(point_on_hull, candidate, p)
            if o < 0:
                # p is more counter‑clockwise
                candidate = p
            elif o == 0:
                # collinear – keep the farthest point to ensure inclusion of intermediate points later
                if (p[0] - point_on_hull[0])**2 + (p[1] - point_on_hull[1])**2 > \

                    candidate = p

        edge_points = []

                continue

                edge_points.append(p)
