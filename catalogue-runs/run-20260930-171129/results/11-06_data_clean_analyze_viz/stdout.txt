import csv
import io
import math
from datetime import datetime
from statistics import mean, median, stdev

CSV_DATA = """Name,Age,Salary,Department,Start Date
Alice,30,75000,Engineering,2020-01-15
Bob,,85000,Engineering,2019-06-01
Charlie,28,,Marketing,2021-03-10

Jack,33,82000,Sales,2020-04-22
,25,70000,Sales,2021-07-18
Leo,38,91000,Engineering,2018-03-01
"""

def parse_csv(data: str):
    reader = csv.DictReader(io.StringIO(data))
    rows = []
    for r in reader:
        rows.append(r)
    return rows

def clean_rows(rows):
    cleaned = []
    for r in rows:
        # Flag missing Name
        if not r["Name"]:
            r["__flag_missing_name"] = True
        else:
            r["__flag_missing_name"] = False
        # Fill missing Age with median age later (placeholder)
        r["Age"] = r["Age"].strip()
        # Salary: convert to float if possible
        salary = r["Salary"].strip()
        try:
            r["Salary"] = float(salary) if salary else None
        except ValueError:
            r["Salary"] = None
        # Department: flag missing
        if not r["Department"]:
            r["__flag_missing_dept"] = True
        else:
            r["__flag_missing_dept"] = False
        # Start Date: parse or flag invalid
        date_str = r["Start Date"].strip()
        try:
            r["Start Date"] = datetime.strptime(date_str, "%Y-%m-%d").date()
            r["__flag_invalid_date"] = False
        except ValueError:
            r["Start Date"] = None
            r["__flag_invalid_date"] = True
        cleaned.append(r)
    # Fill missing Age with median of available ages
    ages = [int(r["Age"]) for r in cleaned if r["Age"]]
    median_age = median(ages) if ages else 0
    for r in cleaned:
        if not r["Age"]:
            r["Age"] = median_age
        else:
            r["Age"] = int(r["Age"])
    return cleaned

def compute_department_stats(rows):
    dept_groups = {}
    for r in rows:
        dept = r["Department"] or "UNKNOWN"
        dept_groups.setdefault(dept, []).append(r)
    stats = []
    for dept, members in dept_groups.items():
        salaries = [m["Salary"] for m in members if m["Salary"] is not None]
        cnt = len(members)
        avg = mean(salaries) if salaries else 0
        med = median(salaries) if salaries else 0
        sd = stdev(salaries) if len(salaries) > 1 else 0
        stats.append({
            "Department": dept,
            "Count": cnt,
            "Mean Salary": round(avg, 2),
            "Median Salary": round(med, 2),
            "Salary StdDev": round(sd, 2)
        })
    return stats

def compute_tenure(rows, ref_date=datetime(2026, 9, 28).date()):
    for r in rows:
        if r["Start Date"]:
            delta = ref_date - r["Start Date"]
            r["Tenure Years"] = round(delta.days / 365.25, 2)
        else:
            r["Tenure Years"] = None
    return rows

def prepare_chart_data(dept_stats):
    # Data for a bar chart of average salary by department
    return [{"Department": d["Department"], "Average Salary": d["Mean Salary"]} for d in dept_stats]

def main():
    rows = parse_csv(CSV_DATA)
    rows = clean_rows(rows)
    rows = compute_tenure(rows)
    dept_stats = compute_department_stats(rows)
    chart_data = prepare_chart_data(dept_stats)
    # Output summary table
    print("Per-Department Statistics:")
    for s in dept_stats:
        print(s)
    print("\nEmployee Tenure:")
    for r in rows:
        print({"Name": r["Name"] or "<missing>", "Tenure Years": r["Tenure Years"]})
    print("\nChart Data (Average Salary by Department):")
    for c in chart_data:
        print(c)

if __name__ == "__main__":
    main()
