def merge_sorted_lists(list1, list2):
    """Merge two sorted lists into a new sorted list.

    Args:
        list1 (list): First sorted list.
        list2 (list): Second sorted list.

    Returns:
        list: A new list containing all elements of ``list1`` and ``list2`` in sorted order.

    Time Complexity:
        O(m + n) where m and n are the lengths of the input lists.
    """
    i, j = 0, 0
    merged = []
    while i < len(list1) and j < len(list2):
        if list1[i] <= list2[j]:
            merged.append(list1[i])
            i += 1
        else:
            merged.append(list2[j])
            j += 1
    # Append any remaining elements
    merged.extend(list1[i:])
    merged.extend(list2[j:])
    return merged

def binary_search_recursive(arr, target, lo=0, hi=None):
    """Recursively search ``target`` in a sorted list ``arr``.

    Args:
        arr (list): Sorted list of comparable items.
        target: Item to locate.
        lo (int, optional): Lower index bound. Defaults to 0.
        hi (int, optional): Upper index bound. Defaults to ``len(arr) - 1``.

    Returns:
        int: Index of ``target`` if found; otherwise -1.

    Time Complexity:
        O(log n) where n = len(arr).
    """
    if hi is None:
        hi = len(arr) - 1
    if lo > hi:
        return -1
    mid = (lo + hi) // 2
    if arr[mid] == target:
        return mid
    elif arr[mid] < target:
        return binary_search_recursive(arr, target, mid + 1, hi)
    else:
        return binary_search_recursive(arr, target, lo, mid - 1)

def flatten(nested):
    """Flatten an arbitrarily nested list structure.

    Args:
        nested (list): List that may contain other lists recursively.

    Returns:
        list: A flat list containing all non‑list elements in the original order.

    Time Complexity:
        O(k) where k is the total number of atomic elements.
    """
    flat = []
    for item in nested:
        if isinstance(item, list):
            flat.extend(flatten(item))
        else:
            flat.append(item)
    return flat
