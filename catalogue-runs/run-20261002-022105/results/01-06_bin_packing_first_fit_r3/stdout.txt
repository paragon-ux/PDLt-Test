def first_fit(items, C):
    bins = []
    for item in items:
        placed = False
        for b in bins:
            if sum(b) + item <= C:
                b.append(item)
                placed = True
                break
        if not placed:
            bins.append([item])
    return bins

def optimal_pack(items, C):
    # simple brute-force search for minimal bins using backtracking
    best = None
    def backtrack(idx, bins):
        nonlocal best
        if best is not None and len(bins) >= len(best):
            return
        if idx == len(items):
            best = [list(b) for b in bins]
            return
        item = items[idx]
        for i, b in enumerate(bins):
            if sum(b) + item <= C:
                b.append(item)
                backtrack(idx+1, bins)
                b.pop()
        # try new bin
        bins.append([item])
        backtrack(idx+1, bins)
        bins.pop()
    backtrack(0, [])
    return best
