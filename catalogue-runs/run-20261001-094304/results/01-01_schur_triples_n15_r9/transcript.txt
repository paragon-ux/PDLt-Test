def find_partition(nums):
    nums_set = set(nums)
    # generate all possible triples (a,b,c) with a+b=c and all distinct
    triples = []
    n = len(nums)
    for i in range(n):
        a = nums[i]
        for j in range(i+1, n):
            b = nums[j]
            c = a + b
            if c in nums_set and c != a and c != b:
                # ensure distinct indices
                if c not in (a, b):
                    triples.append((a, b, c))
    # backtracking exact cover
    used = set()
    solution = []
    triples_by_element = {}
    for t in triples:
        for x in t:
            triples_by_element.setdefault(x, []).append(t)
    def backtrack():
        if len(solution) == 15:
            return True
        # pick unused element with fewest candidate triples
        candidates = [x for x in nums if x not in used]
        if not candidates:
            return False
        element = min(candidates, key=lambda e: len([t for t in triples_by_element.get(e, []) if all(v not in used for v in t)]))
        for t in triples_by_element.get(element, []):

                continue
            solution.append(t)
            for v in t:
                used.add(v)
            if backtrack():
                return True
            for v in t:
                used.remove(v)
            solution.pop()
        return False
    if backtrack():
        return solution
    return None
