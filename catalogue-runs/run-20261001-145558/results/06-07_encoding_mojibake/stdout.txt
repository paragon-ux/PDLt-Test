# Diagnosis and Fix
# The original `save_to_file` writes using `latin-1` while `load_from_file` reads using `utf-8`.
# Characters outside Latin‑1 (e.g., many CJK glyphs and many emoji) cannot be encoded in latin‑1 and are replaced with
# the Unicode replacement character U+FFFD during write, producing garbled output (mojibake).  Even characters that are in
# Latin‑1 but have different byte representations (e.e. é -> 0xE9 in latin‑1 vs. 0xC3 0xA9 in UTF‑8) become corrupted when read
# as UTF‑8.  The fix is to use the same encoding for both operations – UTF‑8 – which can represent all Unicode code points.

def save_to_file(text: str, filepath: str) -> None:
    """Write *text* to *filepath* using UTF‑8 encoding.

    Args:
        text: The Unicode string to persist.
        filepath: Destination file path.
    """
    # The default `encoding='utf-8'` is explicit for clarity.
    with open(filepath, "w", encoding="utf-8") as f:
        f.write(text)


def load_from_file(filepath: str) -> str:
    """Read and return the contents of *filepath* using UTF‑8 encoding.

    Args:
        filepath: Path to the file written by ``save_to_file``.
    """
    with open(filepath, "r", encoding="utf-8") as f:
        return f.read()


def _test_roundtrip() -> None:
    """Test that a string containing Latin, CJK, and emoji round‑trips correctly.

    The test writes the string to a temporary file and reads it back, asserting
    that the original and recovered strings are identical.
    """
    import tempfile
    import os

    #   Latin (Basic Latin & Latin‑1 Supplement): "Café résumé"
