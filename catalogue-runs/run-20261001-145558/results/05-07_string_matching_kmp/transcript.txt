# KMP implementation with detailed trace for pattern "ABABCABAB"

def build_failure_function(pattern: str):
    """Builds the failure function (partial match table) for the given pattern.
    Returns a list where each index i contains the length of the longest proper
    prefix which is also a suffix for pattern[:i+1].
    """
    m = len(pattern)
    failure = [0] * m
    # Trace output
    trace = []
    # The first character always has failure value 0
    trace.append(f"i=0, failure[0]=0")
    j = 0  # length of previous longest prefix suffix
    for i in range(1, m):
        # When mismatch, fall back to the previous longest prefix suffix
        while j > 0 and pattern[i] != pattern[j]:
            trace.append(f"i={i}, j={j} (mismatch), fall back to failure[{j-1}]={failure[j-1]}")
            j = failure[j-1]
        if pattern[i] == pattern[j]:
            j += 1
            failure[i] = j
            trace.append(f"i={i}, match, j incremented to {j}, failure[{i}]={j}")
        else:
            failure[i] = 0
            trace.append(f"i={i}, no match, j stays {j}, failure[{i}]=0")
    # Print the trace
    print("Failure function construction trace:")
    for line in trace:
        print(line)
    print(f"Failure function table: {failure}")
    return failure

def kmp_search(text: str, pattern: str):
    """Searches for all occurrences of pattern in text using KMP.
    Returns a list of starting indices (0‑based).
    """
    n, m = len(text), len(pattern)
    if m == 0:
        return []
    failure = build_failure_function(pattern)
    result = []
    i = j = 0  # i -> index in text, j -> index in pattern
    while i < n:
        if text[i] == pattern[j]:
            i += 1
            j += 1
            if j == m:
                # Full match found
                start_idx = i - m
                result.append(start_idx)
                print(f"Match found at index {start_idx}")
                j = failure[j-1]  # Continue searching for next possible match
        else:
            if j != 0:
                print(f"Mismatch at text[{i}] and pattern[{j}], fallback to failure[{j-1}]={failure[j-1]}")
                j = failure[j-1]
            else:
                i += 1
    return result

if __name__ == "__main__":
    text = "ABABDAABABCABABABABCABAB"
    pattern = "ABABCABAB"
    matches = kmp_search(text, pattern)
    print("All match starting indices:", matches)
