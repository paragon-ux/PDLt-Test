# Cache-Control Header Strategy
# Resource Types and Header Specifications
# ------------------------------------------------------------
# 1. Static assets (JS, CSS, images) – change only on deploy
STATIC_HEADERS = {
    "Cache-Control": "public, max-age=31536000, immutable",
    "Vary": "",
    "ETag": "W/\"static-{HASH}\"",  # weak ETag based on content hash
    "Surrogate-Control": "max-age=31536000, stale-while-revalidate=86400"
}
# 2. User profile data – changes rarely (≤ once per day)
USER_PROFILE_HEADERS = {
    "Cache-Control": "public, max-age=86400, stale-while-revalidate=3600",
    "Vary": "Accept-Encoding, Authorization",
    "ETag": "\"profile-{USER_ID}-{HASH}\"",
    "Surrogate-Control": "max-age=86400, stale-while-revalidate=3600"
}
# 3. Real-time data – must never be stale
REALTIME_HEADERS = {
    "Cache-Control": "no-store, no-cache, must-revalidate",
    "Vary": "",
    "ETag": "",
    "Surrogate-Control": "no-store"
}
# 4. Public catalog data – same for all users, cacheable
PUBLIC_CATALOG_HEADERS = {
    "Cache-Control": "public, max-age=3600, stale-while-revalidate=300",
    "Vary": "",
    "ETag": "\"catalog-{HASH}\"",
    "Surrogate-Control": "max-age=3600, stale-while-revalidate=300"
}
# 5. Authenticated responses – vary by user, private
AUTHENTICATED_HEADERS = {
    "Cache-Control": "private, max-age=0, must-revalidate",
    "Vary": "Authorization",
    "ETag": "\"auth-{USER_ID}-{HASH}\"",
    "Surrogate-Control": "private, max-age=0"
}

# Middleware implementation (compatible with frameworks like FastAPI, Flask, Django)
# The middleware expects the response object to have an attribute `resource_type`
# indicating which category the content belongs to. This attribute should be set
# by the view/controller before the middleware runs.

from typing import Callable, Any

def cache_control_middleware(app: Any) -> Callable[[Any, Any], Any]:
    """Wrap a WSGI-like application with cache‑control header handling.

    The wrapped ``app`` is called with ``environ`` and ``start_response``. After
    obtaining the original response, the middleware inspects ``environ`` for a
    ``resource_type`` key (e.g. ``"static"``, ``"profile"``, ``"realtime"``,
    ``"catalog"``, ``"auth"``) and applies the corresponding header set.
    """

    def middleware(environ, start_response):
        # Call the downstream app to get the response iterable and status
        response_iterable = app(environ, start_response)
        # Gather headers from the start_response call (captured via closure)
        # For simplicity, we assume ``start_response`` stores headers in a list
        # attached to the environ under ``_headers`` – this is a common pattern.
        # If not present, we initialise it.
        headers = environ.get("_headers", [])

        # Determine resource type – default to "static" for safety
        resource_type = environ.get("resource_type", "static")

        # Map resource types to header dictionaries defined above
        header_map = {
            "static": STATIC_HEADERS,
            "profile": USER_PROFILE_HEADERS,
            "realtime": REALTIME_HEADERS,
            "catalog": PUBLIC_CATALOG_HEADERS,
            "auth": AUTHENTICATED_HEADERS,
        }
        selected = header_map.get(resource_type, STATIC_HEADERS)

        # Apply each header; empty values are omitted
        for name, value in selected.items():
            if value:
                # Remove any existing header of the same name first
                headers = [(k, v) for k, v in headers if k.lower() != name.lower()]
                headers.append((name, value))

        # Store back the modified headers for the downstream framework to emit
        environ["_headers"] = headers
        return response_iterable

    return middleware
