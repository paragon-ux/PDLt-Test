import csv
import io
import math
from datetime import datetime
from statistics import mean, median, stdev

csv_data = """Name,Age,Salary,Department,Start Date
Alice,30,75000,Engineering,2020-01-15
Bob,,85000,Engineering,2019-06-01
Charlie,28,,Marketing,2021-03-10
Diana,35,92000,Engineering,2018-11-20
Eve,42,78000,,2017-05-30
Frank,29,71000,Marketing,invalid-date
Grace,31,88000,Engineering,2020-08-12
Hank,27,65000,Marketing,2022-01-05
Ivy,,95000,Engineering,2019-09-15
Jack,33,82000,Sales,2020-04-22
,25,70000,Sales,2021-07-18
Leo,38,91000,Engineering,2018-03-01
"""

# Load CSV
rows = []
reader = csv.DictReader(io.StringIO(csv_data))
for r in reader:
    rows.append(r)

# Clean data
clean_rows = []
for r in rows:
    # Name
    name = r["Name"].strip() if r["Name"].strip() else None
    # Age
    age = int(r["Age"]) if r["Age"].strip().isdigit() else None
    # Salary
    salary = float(r["Salary"]) if r["Salary"].strip().replace('.','',1).isdigit() else None
    # Department
    dept = r["Department"].strip() if r["Department"].strip() else None
    # Start Date
    try:
        start = datetime.strptime(r["Start Date"].strip(), "%Y-%m-%d")
    except Exception:
        start = None
    clean_rows.append({
        "Name": name,
        "Age": age,
        "Salary": salary,
        "Department": dept,
        "Start Date": start,
    })

# Compute tenure
reference_date = datetime.strptime("2026-09-28", "%Y-%m-%d")
for r in clean_rows:
    if r["Start Date"]:
        delta = reference_date - r["Start Date"]
        r["Tenure Years"] = round(delta.days/365.25, 2)
    else:
        r["Tenure Years"] = None

# Per-department statistics
stats = {}
for r in clean_rows:
    dept = r["Department"]
    if not dept:
        continue
    if dept not in stats:
        stats[dept] = []
    if r["Salary"] is not None:
        stats[dept].append(r["Salary"])

summary = []
for dept, salaries in stats.items():
    cnt = len(salaries)
    avg = mean(salaries) if salaries else None
    med = median(salaries) if salaries else None
    std = stdev(salaries) if len(salaries) > 1 else 0.0
    summary.append({
        "Department": dept,
        "Count": cnt,
        "Mean Salary": round(avg,2) if avg is not None else None,
        "Median Salary": round(med,2) if med is not None else None,
        "Salary StdDev": round(std,2)
    })

# Output summary table
print("Summary Statistics per Department:")
for row in summary:
    print(row)

# Data for bar chart (average salary by department)
bar_chart_data = {row["Department"]: row["Mean Salary"] for row in summary}
print("\nBar Chart Data (Avg Salary by Dept):")
print(bar_chart_data)
