import datetime
import re
from typing import List, Set

# Helper functions to parse cron fields
def _parse_field(field: str, min_val: int, max_val: int) -> Set[int]:
    """Parse a single cron field into a set of allowed integer values.
    Supports '*', '*/step', 'a-b', 'a-b/step', and comma‑separated lists.
    """
    result: Set[int] = set()
    if field == '*':
        return set(range(min_val, max_val + 1))
    for part in field.split(','):
        # step syntax
        if '/' in part:
            base, step_s = part.split('/')
            step = int(step_s)
        else:
            base, step = part, 1
        # range or single value
        if base == '*':
            start, end = min_val, max_val
        elif '-' in base:
            start_s, end_s = base.split('-')
            start, end = int(start_s), int(end_s)
        else:
            start = end = int(base)
        # clamp to allowed range
        start = max(start, min_val)
        end = min(end, max_val)
        result.update(range(start, end + 1, step))
    return result

# Cron expression parser
class CronExpression:
    def __init__(self, expr: str):
        fields = expr.strip().split()
        if len(fields) != 5:
            raise ValueError('Cron expression must have exactly 5 fields')
        minute, hour, dom, month, dow = fields
        self.minutes = _parse_field(minute, 0, 59)
        self.hours = _parse_field(hour, 0, 23)
        self.dom = _parse_field(dom, 1, 31)
        self.months = _parse_field(month, 1, 12)
        self.dow = _parse_field(dow, 0, 6)  # 0=Sunday

    def matches(self, dt: datetime.datetime) -> bool:
        return (
            dt.minute in self.minutes and
            dt.hour in self.hours and
            dt.day in self.dom and
            dt.month in self.months and
            dt.weekday() % 7 in self.dow
        )

    def next(self, after: datetime.datetime) -> datetime.datetime:
        # Start searching from the next minute
        nxt = after.replace(second=0, microsecond=0) + datetime.timedelta(minutes=1)
        while True:
            if (nxt.month in self.months and
                nxt.day in self.dom and
                nxt.weekday() % 7 in self.dow and
                nxt.hour in self.hours and
                nxt.minute in self.minutes):
                return nxt
            nxt += datetime.timedelta(minutes=1)

# Public API
def matches(cron_expr: str, dt: datetime.datetime) -> bool:
    """Return True if *dt* matches the cron expression *cron_expr*.
    """
    return CronExpression(cron_expr).matches(dt)

def next_fire(cron_expr: str, after_dt: datetime.datetime) -> datetime.datetime:
    """Return the next datetime after *after_dt* that matches *cron_expr*.
    """
    return CronExpression(cron_expr).next(after_dt)
