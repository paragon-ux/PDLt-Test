from itertools import product

C = 10
items = [5, 5, 5, 3, 3, 3, 7, 7]

def first_fit(items, capacity):
    bins = []
    for item in items:
        placed = False
        for b in bins:
            if sum(b) + item <= capacity:
                b.append(item)
                placed = True
                break
        if not placed:
            bins.append([item])
    return bins

def optimal_packing(items, capacity):
    # backtracking search for minimal number of bins
    best = None
    n = len(items)
    # sort items descending to improve pruning
    items_sorted = sorted(items, reverse=True)
    def backtrack(idx, bins):
        nonlocal best
        if best is not None and len(bins) >= len(best):
            return
        if idx == n:
            best = [list(b) for b in bins]
            return
        item = items_sorted[idx]
        # try to fit into existing bins
        for i, b in enumerate(bins):
            if sum(b) + item <= capacity:
                b.append(item)
                backtrack(idx+1, bins)
                b.pop()
        # or open new bin
        bins.append([item])
        backtrack(idx+1, bins)
        bins.pop()
    backtrack(0, [])
    return best

def verify_packing(packing, items, capacity):
    # check all items used exactly once
    flat = [i for bin_ in packing for i in bin_]
    if sorted(flat) != sorted(items):
        return False
    # check capacity
    for bin_ in packing:
        if sum(bin_) > capacity:
            return False
    return True
