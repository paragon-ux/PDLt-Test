# bookmarks_server.py
import json
import threading
from http.server import BaseHTTPRequestHandler, HTTPServer
from urllib.parse import urlparse, parse_qs
import datetime

# In‑memory storage
bookmarks = {}
next_id = 1

def make_timestamp():
    return datetime.datetime.utcnow().isoformat() + "Z"

class BookmarkHandler(BaseHTTPRequestHandler):
    def _set_headers(self, code=200, content_type="application/json"):
        self.send_response(code)
        self.send_header("Content-Type", content_type)
        self.end_headers()

    def _read_body(self):
        length = int(self.headers.get("Content-Length", 0))
        return self.rfile.read(length).decode("utf-8") if length else ""

    def do_POST(self):
        # Create a new bookmark: POST /bookmarks
        if self.path != "/bookmarks":
            self._set_headers(404)
            self.wfile.write(json.dumps({"error": "Not found"}).encode())
            return
        data = json.loads(self._read_body())
        global next_id
        bookmark = {
            "id": next_id,
            "url": data["url"],
            "title": data.get("title", ""),
            "tags": data.get("tags", []),
            "created_at": make_timestamp()
        }
        bookmarks[next_id] = bookmark
        next_id += 1
        self._set_headers(201)
        self.wfile.write(json.dumps(bookmark).encode())

    def do_GET(self):
        # Retrieve a bookmark or list/search
        parsed = urlparse(self.path)
        if parsed.path == "/bookmarks":
            # list or search by tag
            qs = parse_qs(parsed.query)
            tag = qs.get("tag", [None])[0]
            result = list(bookmarks.values())
            if tag:
                result = [b for b in result if tag in b.get("tags", [])]
            self._set_headers(200)
            self.wfile.write(json.dumps(result).encode())
            return
        elif parsed.path.startswith("/bookmarks/"):
            try:
                bid = int(parsed.path.split("/")[2])
                bookmark = bookmarks[bid]
                self._set_headers(200)
                self.wfile.write(json.dumps(bookmark).encode())
            except (ValueError, KeyError):
                self._set_headers(404)
                self.wfile.write(json.dumps({"error": "Not found"}).encode())
            return
        else:
            self._set_headers(404)
            self.wfile.write(json.dumps({"error": "Not found"}).encode())

    def do_PUT(self):
        # Update a bookmark: PUT /bookmarks/<id>
        if not self.path.startswith("/bookmarks/"):
            self._set_headers(404)
            self.wfile.write(json.dumps({"error": "Not found"}).encode())
            return
        try:
            bid = int(self.path.split("/")[2])
            if bid not in bookmarks:
                raise KeyError
        except (ValueError, KeyError):
            self._set_headers(404)
            self.wfile.write(json.dumps({"error": "Not found"}).encode())
            return
        data = json.loads(self._read_body())
        bookmark = bookmarks[bid]
        # allow partial update
        bookmark.update({
            "url": data.get("url", bookmark["url"]),
            "title": data.get("title", bookmark["title"]),
            "tags": data.get("tags", bookmark["tags"]),
        })
        self._set_headers(200)
        self.wfile.write(json.dumps(bookmark).encode())

    def do_DELETE(self):
        # Delete a bookmark: DELETE /bookmarks/<id>
        if not self.path.startswith("/bookmarks/"):
            self._set_headers(404)
            self.wfile.write(json.dumps({"error": "Not found"}).encode())
            return
        try:
            bid = int(self.path.split("/")[2])
            del bookmarks[bid]
            self._set_headers(204)
            self.wfile.write(b"")
        except (ValueError, KeyError):
            self._set_headers(404)
            self.wfile.write(json.dumps({"error": "Not found"}).encode())

def run_server(port=8000):
    server = HTTPServer(("localhost", port), BookmarkHandler)
    thread = threading.Thread(target=server.serve_forever, daemon=True)
    thread.start()
    return server, thread

if __name__ == "__main__":
    srv, th = run_server()
    print(f"Server running on http://localhost:8000")
    th.join()

# test_bookmarks.py
import json
import urllib.request
import urllib.error
import time
import threading

BASE = "http://localhost:8000"

def http(method, path, data=None, params=None):
    url = BASE + path
    if params:
        query = urllib.parse.urlencode(params)
        url += "?" + query
    req = urllib.request.Request(url, method=method)
    if data is not None:
        encoded = json.dumps(data).encode()
        req.add_header("Content-Type", "application/json")
        req.data = encoded
    try:
        with urllib.request.urlopen(req) as resp:
            body = resp.read().decode()
            return resp.getcode(), json.loads(body) if body else None
    except urllib.error.HTTPError as e:
        return e.code, json.loads(e.read().decode())

def run_tests():
    # start server in background
    import subprocess, sys, os
    # launch the server script
    server_process = subprocess.Popen([sys.executable, "bookmarks_server.py"], stdout=subprocess.PIPE, stderr=subprocess.PIPE)
    time.sleep(0.5)  # give it time to start
    try:
        # CREATE
        status, created = http("POST", "/bookmarks", {"url": "https://example.com", "title": "Example", "tags": ["test"]})
        assert status == 201, f"Create failed: {status}"
        bid = created["id"]
        # READ
        status, read = http("GET", f"/bookmarks/{bid}")
        assert status == 200 and read["url"] == "https://example.com"
        # UPDATE
        status, updated = http("PUT", f"/bookmarks/{bid}", {"title": "Updated"})
        assert status == 200 and updated["title"] == "Updated"
        # SEARCH BY TAG
        status, found = http("GET", "/bookmarks", params={"tag": "test"})
        assert status == 200 and any(b["id"] == bid for b in found)
        # DELETE
        status, _ = http("DELETE", f"/bookmarks/{bid}")
        assert status == 204
        # VERIFY DELETION
        status, _ = http("GET", f"/bookmarks/{bid}")
        assert status == 404
        print("All tests passed")
    finally:
        server_process.terminate()
        server_process.wait()

if __name__ == "__main__":
    run_tests()
