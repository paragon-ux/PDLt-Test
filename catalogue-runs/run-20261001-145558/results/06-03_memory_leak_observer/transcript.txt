class EventEmitter:
    def __init__(self):
        self._listeners = {}

    def on(self, event, callback):
        self._listeners.setdefault(event, []).append(callback)
        return callback  # return the callback for possible removal

    def off(self, event, callback):
        """Unsubscribe a previously subscribed callback from an event."""
        listeners = self._listeners.get(event)
        if listeners and callback in listeners:
            listeners.remove(callback)
            if not listeners:
                del self._listeners[event]

    def emit(self, event, *args):
        for cb in list(self._listeners.get(event, [])):
            cb(*args)

class DataProcessor:
    def __init__(self, emitter):
        self.emitter = emitter
        self.results = []
        # keep a reference to the bound method so we can unsubscribe later
        self._listener = self.emitter.on('data', self.handle_data)

    def handle_data(self, value):
        self.results.append(value * 2)

    def cleanup(self):
        """Remove the listener to allow GC of this instance."""
        self.emitter.off('data', self._listener)
        # break reference cycles
        self._listener = None
        self.emitter = None

def process_batch(emitter, batch):
    processor = DataProcessor(emitter)
    for item in batch:
        emitter.emit('data', item)
    # before returning, detach the listener so the processor can be GC'ed
    processor.cleanup()
    return processor.results
