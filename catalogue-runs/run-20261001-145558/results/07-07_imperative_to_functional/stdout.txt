from functools import reduce
from collections import defaultdict

# Tax rates as given in the original imperative version
_TAX_RATES = {'US': 0.08, 'EU': 0.20, 'UK': 0.20, 'JP': 0.10}
_DEFAULT_TAX_RATE = 0.15


def process_sales_functional(records):
    """Process sales records functionally.

    Steps:
    1. Filter out refunds and non‑positive amounts.
    2. Apply regional tax rates and compute total.
    3. Group records by region.
    4. Summarise count and total per region.
    """
    # Step 1: filter
    valid = list(filter(lambda r: r['amount'] > 0 and r['status'] != 'refunded', records))

    # Step 2: apply tax
    def add_tax(r):
        rate = _TAX_RATES.get(r['region'], _DEFAULT_TAX_RATE)
        total = round(r['amount'] * (1 + rate), 2)
        # return a new dict with extra fields
        return {**r, 'total': total, 'tax_rate': rate}

    taxed = list(map(add_tax, valid))

    # Step 3: group by region using reduce
    def group_by(acc, r):
        acc[r['region']].append(r)
        return acc

    groups = reduce(group_by, taxed, defaultdict(list))

    # Step 4: compute summary per region
    summary = {
        region: {
            'count': len(items),
            'total': round(sum(item['total'] for item in items), 2)
        }
        for region, items in groups.items()
    }
    return summary
