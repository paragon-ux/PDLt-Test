# Subset sum finder for target 40
S = [3, 34, 7, 12, 5, 26, 11, 8, 15, 2, 19, 21]
T = 40
solutions = []

def backtrack(start, current, current_sum):
    if current_sum == T:
        solutions.append(list(current))
        return
    if current_sum > T:
        return
    for i in range(start, len(S)):
        # include S[i]
        current.append(S[i])
        backtrack(i + 1, current, current_sum + S[i])
        current.pop()

backtrack(0, [], 0)
# Emit each solution and verify its sum
