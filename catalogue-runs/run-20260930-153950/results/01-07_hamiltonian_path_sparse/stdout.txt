import sys

def find_hamiltonian_path(adj, n):
    path = []
    visited = [False]*n
    def backtrack(node):
        path.append(node)
        visited[node]=True
        if len(path)==n:
            return True
        for neigh in adj[node]:
            if not visited[neigh]:
                if backtrack(neigh):
                    return True
        visited[node]=False
        path.pop()
        return False
    for start in range(n):
        if backtrack(start):
            return path
    return None
