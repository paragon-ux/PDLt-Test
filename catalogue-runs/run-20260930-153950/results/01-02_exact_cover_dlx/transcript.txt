"""Exact cover solver using Knuth's Algorithm X with dancing links.
Sets S1‑S9 for universe U = {1..9} are hard‑coded. The script searches for all
exact covers and prints them. It also runs a self‑contained test that verifies
each solution covers every element exactly once.
"""

from typing import List, Dict, Set, Tuple

# ---------- Data definition ----------
U = set(range(1, 10))
sets = {
    'S1': {1, 2, 3},
    'S2': {4, 5, 6},
    'S3': {7, 8, 9},
    'S4': {1, 4, 7},
    'S5': {2, 5, 8},
    'S6': {3, 6, 9},
    'S7': {1, 5, 9},
    'S8': {2, 6, 7},
    'S9': {3, 4, 8},
}

# ---------- Dancing‑links structures ----------
class DLXNode:
    __slots__ = ('left', 'right', 'up', 'down', 'column', 'row_name')
    def __init__(self):
        self.left = self.right = self.up = self.down = self
        self.column = None
        self.row_name = None

class ColumnNode(DLXNode):
    __slots__ = ('size', 'name')
    def __init__(self, name: str):
        super().__init__()
        self.size = 0
        self.name = name
        self.column = self

def build_dlx(matrix: Dict[str, Set[int]]) -> Tuple[ColumnNode, Dict[str, List[DLXNode]]]:
    header = ColumnNode('header')
    # create column objects for each element in U
    column_by_elem: Dict[int, ColumnNode] = {}
    last = header
    for elem in sorted(U):
        col = ColumnNode(str(elem))
        column_by_elem[elem] = col
        # link horizontally
        last.right = col
        col.left = last
        last = col
    last.right = header
    header.left = last

    # store nodes per row for later extraction
    row_nodes: Dict[str, List[DLXNode]] = {}

    for row_name, elems in matrix.items():
        first_node = None
        prev_node = None
        for e in elems:
            col = column_by_elem[e]
            node = DLXNode()
            node.column = col
            node.row_name = row_name
            # vertical link
            node.down = col
            node.up = col.up
            col.up.down = node
            col.up = node
            col.size += 1
            # horizontal link within the row
            if first_node is None:
                first_node = node
            else:
                prev_node.right = node
                node.left = prev_node
            prev_node = node
        # close the row circularly
        if first_node is not None:
            first_node.left = prev_node
            prev_node.right = first_node
            row_nodes[row_name] = []
            # collect nodes for this row
            cur = first_node
            while True:
                row_nodes[row_name].append(cur)
                cur = cur.right
                if cur is first_node:
                    break
    return header, row_nodes

# ---------- Algorithm X core ----------
solutions: List[List[str]] = []

def cover(col: ColumnNode):
    col.right.left = col.left
    col.left.right = col.right
    i = col.down
    while i is not col:
        j = i.right
        while j is not i:
            j.down.up = j.up
            j.up.down = j.down
            j.column.size -= 1
            j = j.right
        i = i.down

def uncover(col: ColumnNode):
    i = col.up
    while i is not col:
        j = i.left
        while j is not i:
            j.column.size += 1
            j.down.up = j
            j.up.down = j
            j = j.left
        i = i.up
    col.right.left = col
    col.left.right = col

def search(k: int, header: ColumnNode, partial: List[DLXNode]):
    if header.right is header:
        # all columns covered – record solution
        solutions.append([node.row_name for node in partial])
        return
    # choose column with minimal size (heuristic)
    c = header.right
    min_size = c.size
    col = c
    while c is not header:
        if c.size < min_size:
            min_size = c.size
            col = c
        c = c.right
    cover(col)
    r = col.down
    while r is not col:
        partial.append(r)
        j = r.right
        while j is not r:
            cover(j.column)
            j = j.right
        search(k + 1, header, partial)
        # backtrack
        partial.pop()
        j = r.left
        while j is not r:
            uncover(j.column)
            j = j.left
        r = r.down
    uncover(col)

# ---------- Execution ----------
header, _ = build_dlx(sets)
search(0, header, [])

# Print solutions
print('Exact covers found:')
for sol in solutions:
    print(sol)

# ---------- Self‑contained test ----------
def test_solutions():
    for sol in solutions:
        covered: Set[int] = set()
        for row_name in sol:
            covered.update(sets[row_name])
        assert covered == U, f"Solution {sol} does not cover U"
        # ensure each element appears exactly once
        counts: Dict[int, int] = {e:0 for e in U}
        for row_name in sol:
            for e in sets[row_name]:
                counts[e] += 1
        assert all(v == 1 for v in counts.values()), f"Element multiplicity error in {sol}"
    print('All solutions verified successfully.')

if __name__ == '__main__':
    test_solutions()

        "solutions": solutions
    }
