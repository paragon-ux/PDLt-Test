# ---------------------------------------------------
# Build stage: compile and install dependencies
FROM python:3.11-slim AS builder
WORKDIR /app
# Install build‑time dependencies only
RUN apt-get update && apt-get install -y --no-install-recommends \
    gcc libpq-dev curl wget && rm -rf /var/lib/apt/lists/*
# Copy only what is needed to install Python packages
COPY requirements.txt ./
RUN pip install --no-cache-dir -r requirements.txt
# Copy source code (excluding tests) and build the app
COPY . ./
# Remove test files and any unnecessary build artefacts
RUN find . -type d -name "tests" -exec rm -rf {} + \
    && find . -type f -name "*.pyc" -delete
# Build step (if needed) – placeholder for custom build commands
# e.g., python setup.py build (uncomment if required)
# RUN python setup.py build
# ---------------------------------------------------
# Runtime stage: minimal image with only runtime deps
FROM gcr.io/distroless/python3.11-slim AS runtime
WORKDIR /app
# Copy only the installed site‑packages and the application code
COPY --from=builder /usr/local/lib/python3.11/site-packages /usr/local/lib/python3.11/site-packages
COPY --from=builder /app /app
# Expose the application port
EXPOSE 8000
# Command to run the service
CMD ["python", "-m", "uvicorn", "app.main:app", "--host", "0.0.0.0"]

# ---------------------------------------------------
# Explanations of each optimization step
# 1. **Multi‑stage build** – Separates build‑time (builder) and runtime (runtime) stages.
#    This allows us to discard compilers, package managers, and other heavy tools.
# 2. **Slim base for builder** – `python:3.11-slim` is smaller than the full image while still providing
#    the necessary OS tools for compiling native extensions.
# 3. **Distroless runtime** – `gcr.io/distroless/python3.11-slim` contains only the Python runtime and its
#    dependencies, removing the OS package manager, shells, and other utilities, drastically reducing size.
# 4. **Layer caching** – Installing `requirements.txt` before copying the rest of the source ensures that
#    changes to the application code do not invalidate the cached layer of installed Python packages.
# 5. **Excluding tests** – The `find` command removes the `tests` directory and compiled byte‑code files in the
#    final image, preventing unnecessary files from being shipped.
# 6. **Copy only necessary artefacts** – The runtime stage copies only the compiled site‑packages and the application
#    source, omitting the build‑time packages (`gcc`, `curl`, etc.).
# 7. **No apt caches** – `rm -rf /var/lib/apt/lists/*` after installing packages prevents leftover package lists.
# 8. **No intermediate files** – The builder stage deletes temporary files, keeping the final image lean.
# 9. **Port exposure and CMD** – Remain unchanged, defining the service entry point.
