import re
import sys

def markdown_to_html(md: str) -> str:
    lines = md.split('\n')
    html_blocks = []
    i = 0
    in_code_block = False
    code_block_buffer = []
    while i < len(lines):
        line = lines[i]
        # fenced code block start/end
        if line.startswith('```'):
            if in_code_block:
                html_blocks.append('<pre><code>' + '\n'.join(code_block_buffer) + '\n</code></pre>')
                code_block_buffer = []
                in_code_block = False
            else:
                in_code_block = True
            i += 1
            continue
        # indented code block (four spaces)
        if not in_code_block and line.startswith('    '):
            # collect consecutive indented lines
            indented = []
            while i < len(lines) and lines[i].startswith('    '):
                indented.append(lines[i][4:])
                i += 1
            html_blocks.append('<pre><code>' + '\n'.join(indented) + '\n</code></pre>')
            continue
        if in_code_block:
            code_block_buffer.append(line)
            i += 1
            continue
        # header
        header_match = re.match(r'^(#{1,6})\s*(.*)$', line)
        if header_match:
            level = len(header_match.group(1))
            content = header_match.group(2).strip()
            html_blocks.append(f'<h{level}>' + inline_format(content) + f'</h{level}>')
            i += 1
            continue
        # unordered list
        if re.match(r'^\s*[-*]\s+', line):
            items = []
            while i < len(lines) and re.match(r'^\s*[-*]\s+', lines[i]):
                item = re.sub(r'^\s*[-*]\s+', '', lines[i])
                items.append('<li>' + inline_format(item) + '</li>')
                i += 1
            html_blocks.append('<ul>' + ''.join(items) + '</ul>')
            continue
        # blank line -> separate paragraphs
        if line.strip() == '':
            i += 1
            continue
        # paragraph (collect consecutive non-blank lines)
        para_lines = []
        while i < len(lines) and lines[i].strip() != '' and not re.match(r'^(#{1,6})\s', lines[i]) and not lines[i].startswith('```') and not lines[i].startswith('    '):
            para_lines.append(lines[i])
            i += 1
        paragraph = ' '.join(para_lines).strip()
        html_blocks.append('<p>' + inline_format(paragraph) + '</p>')
    return '\n'.join(html_blocks)

def inline_format(text: str) -> str:
    # Escape HTML special chars
    text = (text.replace('&', '&amp;')
                .replace('<', '&lt;')
                .replace('>', '&gt;'))
    # code spans
    text = re.sub(r'`([^`]+?)`', r'<code>\1</code>', text)
    # links
    text = re.sub(r'\[([^\]]+)\]\(([^)]+)\)', r'<a href="\2">\1</a>', text)
    # bold (both ** and __)
    text = re.sub(r'(?<!\*)\*\*([^*]+?)\*\*(?!\*)', r'<strong>\1</strong>', text)
    text = re.sub(r'__(.+?)__', r'<strong>\1</strong>', text)
    # italic (both * and _)
    text = re.sub(r'(?<!\*)\*([^*]+?)\*(?!\*)', r'<em>\1</em>', text)
    text = re.sub(r'_([^_]+?)_', r'<em>\1</em>', text)
    return text

import unittest

class TestMarkdownToHTML(unittest.TestCase):
    def assertHTML(self, md, expected):
        self.assertEqual(markdown_to_html(md).strip(), expected.strip())

    def test_headers(self):
        self.assertHTML('# H1', '<h1>H1</h1>')
        self.assertHTML('## H2', '<h2>H2</h2>')
        self.assertHTML('###### H6', '<h6>H6</h6>')
        self.assertHTML('####### Not a header', '<p>####### Not a header</p>')
        self.assertHTML('# ', '<h1></h1>')

    def test_bold_italic(self):
        self.assertHTML('**bold**', '<p><strong>bold</strong></p>')
        self.assertHTML('__bold__', '<p><strong>bold</strong></p>')
        self.assertHTML('*italic*', '<p><em>italic</em></p>')
        self.assertHTML('_italic_', '<p><em>italic</em></p>')
        self.assertHTML('**bold inside *italic***', '<p><strong>bold inside <em>italic</em></strong></p>')
        self.assertHTML('*italic inside **bold***', '<p><em>italic inside <strong>bold</strong></em></p>')

    def test_inline_code(self):
        self.assertHTML('`code`', '<p><code>code</code></p>')
        self.assertHTML('Mix `code` with text', '<p>Mix <code>code</code> with text</p>')

    def test_links(self):
        self.assertHTML('[example](http://example.com)', '<p><a href="http://example.com">example</a></p>')
        self.assertHTML('Link with special chars [test](https://example.com/?q=1&b=2)', '<p>Link with special chars <a href="https://example.com/?q=1&b=2">test</a></p>')

    def test_unordered_lists(self):
        md = '- a\n- b\n* c'
        expected = '<ul><li>a</li><li>b</li><li>c</li></ul>'
        self.assertHTML(md, expected)

    def test_code_blocks(self):
        md_fenced = '```\ncode line 1\ncode line 2\n```'
        expected_fenced = '<pre><code>code line 1\ncode line 2\n</code></pre>'
        self.assertHTML(md_fenced, expected_fenced)
        md_indent = '    indented code line 1\n    indented code line 2'
        expected_indent = '<pre><code>indented code line 1\nindented code line 2\n</code></pre>'
        self.assertHTML(md_indent, expected_indent)

    def test_paragraphs(self):
        md = 'First line\nSecond line\n\nThird line'
        expected = '<p>First line Second line</p>\n<p>Third line</p>'
        self.assertHTML(md, expected)

    def test_complex(self):
        md = '''# Header\n\nParagraph with **bold**, *italic*, `code` and a [link](https://example.com).\n\n- List item with **bold**\n- Another *italic* item\n\n    def foo():\n        return \'bar\''' 
        expected = '''<h1>Header</h1>\n<p>Paragraph with <strong>bold</strong>, <em>italic</em>, <code>code</code> and a <a href="https://example.com">link</a>.</p>\n<ul><li>List item with <strong>bold</strong></li><li>Another <em>italic</em> item</li></ul>\n<pre><code>def foo():\n    return 'bar'\n</code></pre>''' 
        self.assertHTML(md, expected)
