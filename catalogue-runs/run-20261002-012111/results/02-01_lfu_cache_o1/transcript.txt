    class DoublyLinkedList:
        __slots__ = ('head', 'tail')
        def __init__(self):
            self.head = self.tail = None
        def append(self, node):
            node.prev = node.next = None
            if not self.head:
                self.head = self.tail = node
            else:
                self.tail.next = node
                node.prev = self.tail
                self.tail = node
        def pop_left(self):
            if not self.head:
                return None
            node = self.head
            self.head = node.next
            if self.head:
                self.head.prev = None
            else:
                self.tail = None
            node.prev = node.next = None
            return node
        def remove(self, node):

                node.prev.next = node.next
            else:
                self.head = node.next

                node.next.prev = node.prev
            else:

        def is_empty(self):
