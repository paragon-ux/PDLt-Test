import threading

class PoolExhausted(Exception):
    """Raised when no free blocks are available in the memory pool."""
    pass

class DoubleFreeError(Exception):
    """Raised when an attempt is made to free a block that is already free."""
    pass

class FixedBlockMemoryPool:
    """A fixed‑size block memory pool.

    The pool is backed by a bytearray of size ``N * B`` where ``N`` is the number of
    blocks and ``B`` is the block size in bytes.  A free‑list is embedded inside the
    buffer: each free block stores the offset of the next free block as a 4‑byte
    little‑endian integer.  The head of the free‑list is kept in ``self._free_head``.
    """

    _INT_SIZE = 4  # we store offsets as 4‑byte unsigned ints

    def __init__(self, N: int, B: int):
        if N <= 0 or B <= 0:
            raise ValueError("N and B must be positive integers")
        self.N = N
        self.B = B
        self._buffer = bytearray(N * B)
        # initialise free‑list; each block points to the next block
        for i in range(N):
            offset = i * B
            next_offset = (i + 1) * B if i + 1 < N else -1  # -1 marks end of list
            self._buffer[offset:offset + self._INT_SIZE] = next_offset.to_bytes(self._INT_SIZE, "little", signed=True)
        self._free_head = 0  # start with first block
        self._lock = threading.Lock()
        # keep a set of free offsets for double‑free detection (no extra list allocation after init)
        self._free_set = set(range(0, N * B, B))

    def allocate(self) -> int:
        """Allocate a block and return its offset.

        Raises
        ------
        PoolExhausted
            If no free blocks remain.
        """
        with self._lock:
            if self._free_head == -1:
                raise PoolExhausted("Memory pool exhausted")
            offset = self._free_head
            # read next free offset from the block
            next_bytes = self._buffer[offset:offset + self._INT_SIZE]
            self._free_head = int.from_bytes(next_bytes, "little", signed=True)
            # remove from free‑set
            self._free_set.remove(offset)
            return offset

    def free(self, offset: int) -> None:
        """Return a block to the pool.

        Parameters
        ----------
        offset : int
            The offset previously returned by ``allocate``.

        Raises
        ------
        DoubleFreeError
            If the block at ``offset`` is already free.
        ValueError
            If ``offset`` does not correspond to a block boundary.
        """
        if offset % self.B != 0 or not (0 <= offset < self.N * self.B):
            raise ValueError("Invalid block offset")
        with self._lock:
            if offset in self._free_set:
                raise DoubleFreeError(f"Block at offset {offset} is already free")
            # prepend this block to the free list
            next_bytes = self._free_head.to_bytes(self._INT_SIZE, "little", signed=True)
            self._buffer[offset:offset + self._INT_SIZE] = next_bytes
            self._free_head = offset
            self._free_set.add(offset)

if __name__ == "__main__":
    import unittest

    class TestFixedBlockMemoryPool(unittest.TestCase):
        def setUp(self):
            self.N = 8
            self.B = 16
            self.pool = FixedBlockMemoryPool(self.N, self.B)

        def test_allocate_all_blocks(self):
            offsets = [self.pool.allocate() for _ in range(self.N)]
            self.assertEqual(len(set(offsets)), self.N)  # all distinct
            with self.assertRaises(PoolExhausted):
                self.pool.allocate()

        def test_free_and_reallocate(self):
            offsets = [self.pool.allocate() for _ in range(self.N)]
            # free a few blocks
            self.pool.free(offsets[2])
            self.pool.free(offsets[5])
            # re‑allocate should give the most recently freed block first (LIFO)
            self.assertEqual(self.pool.allocate(), offsets[5])
            self.assertEqual(self.pool.allocate(), offsets[2])
            # now exhausted again
            with self.assertRaises(PoolExhausted):
                self.pool.allocate()

        def test_double_free_detection(self):
            off = self.pool.allocate()
            self.pool.free(off)
            with self.assertRaises(DoubleFreeError):
                self.pool.free(off)

    unittest.main()
