import json

# Define the wheel graph W_11
nodes = list(range(12))  # 0 is hub, 1-11 are outer cycle
edges = []
# Hub edges
for i in range(1, 12):
    edges.append((0, i))
# Outer cycle edges
for i in range(1, 12):
    edges.append((i, i % 11 + 1))

# Colors available
colors = [1, 2, 3, 4]

# Adjacency dictionary
adj = {n: set() for n in nodes}
for u, v in edges:
    adj[u].add(v)
    adj[v].add(u)

# Backtracking search for a valid 4-coloring
solution = {}

def backtrack(node_index):
    if node_index == len(nodes):
        return True
    node = nodes[node_index]
    for c in colors:
        if all(solution.get(nb) != c for nb in adj[node]):
            solution[node] = c
            if backtrack(node_index + 1):
                return True
            del solution[node]
    return False

found = backtrack(0)

        "polarity": "positive",
