class RollbackUnionFind:
    def __init__(self):
        self.parent = {}
        self.rank = {}
        self.history = []  # stack of snapshots

    def make_set(self, x):
        if x in self.parent:
            raise ValueError("Element already exists")
        self.parent[x] = x
        self.rank[x] = 0

    def find(self, x):
        if x not in self.parent:
            raise KeyError("Element not found")
        while self.parent[x] != x:
            x = self.parent[x]
        return x

    def union(self, x, y):
        rx = self.find(x)
        ry = self.find(y)
        if rx == ry:
            return
        # union by rank, record changes for rollback
        if self.rank[rx] < self.rank[ry]:
            self.parent[rx] = ry
        elif self.rank[rx] > self.rank[ry]:
            self.parent[ry] = rx
        else:
            self.parent[ry] = rx
            self.rank[rx] += 1

    def save(self):
        # snapshot current parent and rank dictionaries (shallow copy is enough)
        self.history.append((self.parent.copy(), self.rank.copy()))

    def restore(self):
        if not self.history:
            raise RuntimeError("No saved state to restore")
        self.parent, self.rank = self.history.pop()
