import random
import unittest

class SkipListNode:
    __slots__ = ('key', 'value', 'forward')
    def __init__(self, key, value, level):
        self.key = key
        self.value = value
        self.forward = [None] * (level + 1)

class SkipListMap:
    MAX_LEVEL = 16
    P = 0.5

    def __init__(self):
        self.header = SkipListNode(None, None, self.MAX_LEVEL)
        self.level = 0
        self.length = 0

    def random_level(self):
        lvl = 0
        while random.random() < self.P and lvl < self.MAX_LEVEL:
            lvl += 1
        return lvl

    def insert(self, key, value):
        update = [None] * (self.MAX_LEVEL + 1)
        current = self.header
        for i in range(self.level, -1, -1):
            while current.forward[i] and current.forward[i].key < key:
                current = current.forward[i]
            update[i] = current
        current = current.forward[0]
        if current and current.key == key:
            current.value = value
            return
        lvl = self.random_level()
        if lvl > self.level:
            for i in range(self.level + 1, lvl + 1):
                update[i] = self.header
            self.level = lvl
        new_node = SkipListNode(key, value, lvl)
        for i in range(lvl + 1):
            new_node.forward[i] = update[i].forward[i]
            update[i].forward[i] = new_node
        self.length += 1

    def search(self, key):
        current = self.header
        for i in range(self.level, -1, -1):

                current = current.forward[i]

        return None

class TestSkipListMap(unittest.TestCase):
    def setUp(self):
        self.map = SkipListMap()
        self.keys = list(range(1, 101))
        random.shuffle(self.keys)
        for k in self.keys:
            self.map.insert(k, str(k))

    def test_ordering_invariant(self):
        # In-order traversal via range_query should produce sorted keys
        result = self.map.range_query(1, 100)
        sorted_keys = sorted(self.keys)
        self.assertEqual([k for k, v in result], sorted_keys)
        # Verify search returns correct values
        for k in self.keys:
            self.assertEqual(self.map.search(k), str(k))
        # Delete a few keys and check invariants
        for k in self.keys[:10]:
            self.map.delete(k)
        result_after = self.map.range_query(1, 100)
        remaining_keys = sorted(self.keys[10:])
        self.assertEqual([k for k, v in result_after], remaining_keys)

    def test_level_distribution(self):
        levels = [0] * (self.map.MAX_LEVEL + 1)
        for _ in range(1000):
            node_level = self.map.random_level()
            levels[node_level] += 1
        total = sum(levels)
        # Expect roughly half nodes at level 0, quarter at level 1, etc.
        for lvl, count in enumerate(levels[:5]):  # check first few levels
            expected = total * (0.5 ** (lvl + 1))
            self.assertTrue(abs(count - expected) / total < 0.2)

    def test_range_query(self):
        # Query a subrange
        lo, hi = 20, 30
        result = self.map.range_query(lo, hi)
        expected_keys = [k for k in range(lo, hi + 1) if k in self.keys]
        self.assertEqual([k for k, v in result], expected_keys)
