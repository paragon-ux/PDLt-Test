import threading

counter = 0
counter_lock = threading.Lock()

def increment_with_lock(n):
    global counter
    for _ in range(n):
        # protect the increment operation
        with counter_lock:
            counter += 1

threads = [threading.Thread(target=increment_with_lock, args=(100_000,)) for _ in range(4)]
